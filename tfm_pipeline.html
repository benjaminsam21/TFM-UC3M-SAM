oa_fetch_retry <- function(..., max_tries = 5, base_delay = 2) {
  for (attempt in seq_len(max_tries)) {
    result <- tryCatch(oa_fetch(...), error = function(e) e)
    ok <- !inherits(result, "error") && !is.null(result) &&
      !(is.data.frame(result) && nrow(result) == 0)
    if (ok) return(result)
    if (attempt == max_tries) {
      if (inherits(result, "error")) stop(result)
      stop("oa_fetch_retry: oa_fetch() returned NULL/empty after ", max_tries,
           " attempts (a rate-limit or API error that didn't surface as an R error)")
    }
    msg <- if (inherits(result, "error")) conditionMessage(result) else "oa_fetch() returned NULL/empty"
    wait <- base_delay * 2^(attempt - 1) + stats::runif(1, 0, 1)
    message(sprintf("oa_fetch failed (attempt %d/%d): %s -- retrying in %.1fs",
                     attempt, max_tries, msg, wait))
    Sys.sleep(wait)
  }
}

}

extract_primary_subfield <- function(topics_df) {
  if (is.null(topics_df) || nrow(topics_df) == 0) return(NA_character_)
  row <- topics_df[topics_df$i == 1 & topics_df$type == "subfield", ]

fetch_works_progress <- function(cache_file, checkpoint_label, per_page = 200,
                                  pages_per_chunk = 20, ...) {
  if (file.exists(cache_file)) {
    message("Loading cached works from ", cache_file)
    return(readRDS(cache_file))
  }

  filter_args <- list(...)
  checkpoint_dir <- file.path(data_dir, paste0(checkpoint_label, "_checkpoints"))
  dir.create(checkpoint_dir, showWarnings = FALSE)

  count_result <- do.call(oa_fetch_retry, c(
    list(entity = "works", count_only = TRUE, verbose = FALSE,
         options = list(corpus = "all")),
    filter_args
  ))
  n_records <- extract_count(count_result)
  n_pages <- ceiling(n_records / per_page)
  cat(sprintf("%s: %s records expected across %d pages (per_page=%d)\n",
              checkpoint_label, format(n_records, big.mark = ","), n_pages, per_page))

  if (n_records > 10000) {
    cat(sprintf("%s: %s records exceeds the 10,000-result offset-pagination limit -- doing a single un-chunked cursor fetch (no per-chunk resume for this step)\n",
                checkpoint_label, format(n_records, big.mark = ",")))
    works <- do.call(oa_fetch_retry, c(
      list(entity = "works", per_page = per_page, verbose = FALSE,
           options = list(corpus = "all")),
      filter_args
    ))
    if (is.null(works) || nrow(works) == 0) {
      stop(checkpoint_label, ": got NULL/empty result, refusing to cache it")
    }
    saveRDS(works, cache_file)
    unlink(checkpoint_dir, recursive = TRUE)
    return(works)
  }

  page <- 1
  while (page <= n_pages) {
    page_end <- min(page + pages_per_chunk - 1, n_pages)
    chunk_file <- file.path(checkpoint_dir, sprintf("pages_%05d_%05d.rds", page, page_end))

    if (file.exists(chunk_file)) {
      cat(sprintf("Skipping pages %d-%d (already fetched, resuming)\n", page, page_end))
      page <- page_end + 1
      next
    }

    t0 <- Sys.time()
    cat(sprintf("[%s] %s: fetching pages %d-%d of %d...\n",
                format(t0, "%H:%M:%S"), checkpoint_label, page, page_end, n_pages))

    chunk <- do.call(oa_fetch_retry, c(
      list(entity = "works", per_page = per_page, paging = "page",
           pages = page:page_end, verbose = FALSE,
           options = list(corpus = "all")),
      filter_args
    ))
    elapsed <- round(as.numeric(difftime(Sys.time(), t0, units = "secs")), 1)
    cat(sprintf("  -> got %d rows in %ss\n", nrow(chunk), elapsed))

    saveRDS(chunk, chunk_file)
    page <- page_end + 1
  }

  chunk_files <- list.files(checkpoint_dir, full.names = TRUE)
  works <- bind_rows(lapply(chunk_files, readRDS)) %>% distinct(id, .keep_all = TRUE)
  if (is.null(works) || nrow(works) == 0) {
    stop(checkpoint_label, ": got NULL/empty combined result, refusing to cache it")
  }
  saveRDS(works, cache_file)
  unlink(checkpoint_dir, recursive = TRUE)
  works
}

fetch_core_authors <- function(institution_id, institution_label, from, to) {
  works <- fetch_works_progress(
    cache_file        = cache_path(institution_label, "core_works"),
    checkpoint_label  = paste0(institution_label, "_core_works"),
    "institutions.id"        = institution_id,
    from_publication_date    = from,
    to_publication_date      = to,
    "primary_topic.field.id" = "33"
  )
  authors_flat <- works %>%
    select(id, authorships) %>%
    unnest(authorships, names_sep = "_")
  unique(authors_flat$authorships_id[!is.na(authors_flat$authorships_id)])
}

fetch_broad_works <- function(core_author_ids, institution_label, from, to, batch_size = 50) {
  cache_file <- cache_path(institution_label, "broad_works")
  if (file.exists(cache_file)) {
    message("Loading cached broad-fetch works from ", cache_file)
    return(readRDS(cache_file))
  }

  checkpoint_dir <- file.path(data_dir, paste0(institution_label, "_broad_works_checkpoints"))
  dir.create(checkpoint_dir, showWarnings = FALSE)

  id_batches <- split(core_author_ids, ceiling(seq_along(core_author_ids) / batch_size))
  n_batches <- length(id_batches)
  cat(sprintf("%s_broad_works: %d authors split into %d batches of up to %d\n",
              institution_label, length(core_author_ids), n_batches, batch_size))

  for (i in seq_along(id_batches)) {
    batch_file <- file.path(checkpoint_dir, sprintf("batch_%05d.rds", i))
    if (file.exists(batch_file)) {
      cat(sprintf("Skipping batch %d/%d (already fetched, resuming)\n", i, n_batches))
      next
    }

    t0 <- Sys.time()
    cat(sprintf("[%s] batch %d/%d: fetching works for %d authors...\n",
                format(t0, "%H:%M:%S"), i, n_batches, length(id_batches[[i]])))

    works_batch <- oa_fetch_retry(
      entity                = "works",
      author.id              = id_batches[[i]],
      from_publication_date  = from,
      to_publication_date    = to,
      verbose                = FALSE,
      options                = list(corpus = "all")
    )
    elapsed <- round(as.numeric(difftime(Sys.time(), t0, units = "secs")), 1)
    cat(sprintf("  -> got %d rows in %ss\n", nrow(works_batch), elapsed))

    saveRDS(works_batch, batch_file)
  }

  batch_files <- list.files(checkpoint_dir, full.names = TRUE)
  works <- bind_rows(lapply(batch_files, readRDS)) %>% distinct(id, .keep_all = TRUE)
  if (is.null(works) || nrow(works) == 0) {
    stop("fetch_broad_works: got NULL/empty combined result, refusing to cache it")
  }
  saveRDS(works, cache_file)
  unlink(checkpoint_dir, recursive = TRUE)
  works
}

works_to_edges <- function(works) {
  works <- works %>% distinct(id, .keep_all = TRUE)
  works$primary_subfield <- purrr::map_chr(works$topics, extract_primary_subfield)

  works$authorships <- lapply(works$authorships, function(a) {
    a$institution_ids <- lapply(a$affiliations, function(aff) {
      if (is.null(aff)) character(0) else aff$id
    })
    a[, c("id", "display_name", "orcid", "author_position", "is_corresponding", "institution_ids")]
  })

  works %>%
    select(work_id = id, publication_date, publication_year, primary_subfield,
           cited_by_count, fwci, type, is_retracted, authorships) %>%
    unnest(authorships, names_sep = "_") %>%
    rename(
      author_id              = authorships_id,
      author_name            = authorships_display_name,
      author_orcid            = authorships_orcid,
      author_position        = authorships_author_position,
      is_corresponding       = authorships_is_corresponding,
      author_institution_ids = authorships_institution_ids
    ) %>%
    distinct(work_id, author_id, .keep_all = TRUE)
}

build_network_from_edges <- function(edges, max_authors = Inf, restrict_to_ids = NULL) {
  n_authors_per_work <- edges %>% count(work_id, name = "n_authors")
  keep_works <- n_authors_per_work %>% filter(n_authors <= max_authors) %>% pull(work_id)
  edges <- edges %>% filter(work_id %in% keep_works)

  authors_clean <- edges %>%
    select(paper_id = work_id, author_id, author_name, author_position) %>%
    filter(!is.na(author_id))

  coauthor_edges <- authors_clean %>%
    inner_join(authors_clean, by = "paper_id", relationship = "many-to-many") %>%
    filter(author_id.x < author_id.y)

  if (!is.null(restrict_to_ids)) {
    coauthor_edges <- coauthor_edges %>%
      filter(author_id.x %in% restrict_to_ids, author_id.y %in% restrict_to_ids)
  }

  author_nodes <- if (!is.null(restrict_to_ids)) {
    data.frame(author_id = restrict_to_ids) %>%
      left_join(authors_clean %>% distinct(author_id, author_name), by = "author_id")
  } else {
    authors_clean %>% group_by(author_id) %>% slice(1) %>% ungroup()
  }

  g <- graph_from_data_frame(
    d = coauthor_edges %>%
      select(from = author_id.x, to = author_id.y,
             position_from = author_position.x, position_to = author_position.y),
    directed = FALSE,
    vertices = author_nodes %>% select(author_id, author_name)
  )

  E(g)$weight <- 1
  g <- igraph::simplify(g, remove.multiple = TRUE, remove.loops = TRUE,
                         edge.attr.comb = list(weight = "sum", "ignore"))

  list(g = g, authors_clean = authors_clean, edges = edges)
}

network_diagnostics <- function(g, label) {
  comp <- components(g)
  data.frame(
    label            = label,
    nodes            = vcount(g),
    edges            = ecount(g),
    components       = comp$no,
    largest_comp     = max(comp$csize),
    giant_comp_share = round(max(comp$csize) / vcount(g), 3),
    density          = round(edge_density(g), 4),
    mean_degree      = round(mean(degree(g)), 2),
    isolated         = sum(degree(g) == 0)
  )
}

build_seniority_age <- function(author_profiles_inst, current_year) {
  author_profiles_inst %>%
    select(author_id = id, display_name, h_index, counts_by_year) %>%
    rowwise() %>%
    mutate(
      first_year_seen = min(counts_by_year$year, na.rm = TRUE),
      academic_age     = current_year - first_year_seen + 1,
      m_quotient       = h_index / academic_age
    ) %>%
    ungroup() %>%
    filter(academic_age > 0) %>%
    select(author_id, h_index, academic_age, m_quotient)
}

  build_seniority_age(author_profiles_inst, current_year)

genderize_cache_file <- file.path(data_dir, "genderize_cache.rds")

load_genderize_cache <- function() {
  if (file.exists(genderize_cache_file)) {
    cache <- readRDS(genderize_cache_file)
    if (!"country_id" %in% names(cache)) cache$country_id <- NA_character_
    cache
  } else {
    data.frame(first_name = character(), country_id = character(), gender = character(),
               probability = double(), count = integer(), stringsAsFactors = FALSE)
  }
}

classify_gender <- function(gender, probability, count, prob_threshold = 0.95, min_count = 5) {
  if (is.na(gender)) return("unclassifiable")
  if (is.na(probability) || is.na(count)) return("unclassifiable")
  if (probability < prob_threshold || count < min_count) return("unclassifiable")
  gender
}

extract_first_name <- function(full_name) {
  first <- str_extract(str_trim(full_name), "^\\S+")
  first_clean <- str_remove_all(first, "[.,]")
  if (is.na(first_clean) || nchar(first_clean) <= 1) return(NA_character_)
  first_clean
}

build_authors_with_gender <- function(author_profiles_inst) {
  full_cache <- load_genderize_cache() %>%
    rowwise() %>%
    mutate(name_inferred_gender = classify_gender(gender, probability, count)) %>%
    ungroup()
  gender_lookup_global <- full_cache %>% filter(is.na(country_id))
  gender_lookup_es     <- full_cache %>% filter(country_id == "ES")

  core_authors <- author_profiles_inst %>%
    select(author_id = id, author_name = display_name) %>%
    distinct(author_id, .keep_all = TRUE)

  core_authors %>%
    mutate(first_name = purrr::map_chr(author_name, extract_first_name)) %>%
    left_join(gender_lookup_global %>% select(first_name, name_inferred_gender_global = name_inferred_gender),

fetch_genderize_batch <- function(names_batch, country_id = NULL,
                                   api_key = Sys.getenv("GENDERIZE_API_KEY"),
                                   max_tries = 5, base_delay = 2) {
  qs  <- paste0("name[]=", vapply(names_batch, utils::URLencode, character(1), reserved = TRUE),
                collapse = "&")
  url <- paste0("https://api.genderize.io/?", qs)
  if (!is.null(country_id)) url <- paste0(url, "&country_id=", country_id)
  if (nzchar(api_key)) url <- paste0(url, "&apikey=", api_key)

  for (attempt in seq_len(max_tries)) {
    result <- tryCatch(jsonlite::fromJSON(url), error = function(e) e)
    ok <- !inherits(result, "error") && is.data.frame(result) && "gender" %in% names(result)
    if (ok) return(result)

    if (attempt == max_tries) {
      if (inherits(result, "error")) stop(result)
      stop("fetch_genderize_batch: unexpected response after ", max_tries, " attempts: ",
           paste(names(result), collapse = ", "))
    }
    msg <- if (inherits(result, "error")) conditionMessage(result) else
      paste("unexpected response shape:", paste(names(result), collapse = ", "))
    wait <- base_delay * 2^(attempt - 1)
    message(sprintf("genderize batch failed (attempt %d/%d): %s -- retrying in %ds",
                     attempt, max_tries, msg, wait))
    Sys.sleep(wait)
  }
}

query_genderize_for_names <- function(names_to_check, country_id = NULL, batch_size = 100) {
  genderize_cache <- load_genderize_cache()
  cache_mask <- if (is.null(country_id)) {
    is.na(genderize_cache$country_id)
  } else {
    genderize_cache$country_id == country_id
  }
  already_done <- genderize_cache$first_name[cache_mask]
  to_query <- setdiff(names_to_check, already_done)

  label <- if (is.null(country_id)) "global" else country_id
  cat(sprintf("%s new names to query for country_id = %s (%s already cached)\n",
              length(to_query), label, length(names_to_check) - length(to_query)))
  if (length(to_query) == 0) return(invisible(genderize_cache))

  checkpoint_dir <- file.path(data_dir, paste0("genderize_", label, "_bs", batch_size, "_checkpoints"))
  dir.create(checkpoint_dir, showWarnings = FALSE)

  batches <- split(to_query, ceiling(seq_along(to_query) / batch_size))
  n_batches <- length(batches)

  for (i in seq_along(batches)) {
    batch_file <- file.path(checkpoint_dir, sprintf("batch_%05d.rds", i))
    if (file.exists(batch_file)) {
      cat(sprintf("Skipping batch %d/%d (already fetched, resuming)\n", i, n_batches))
      next
    }

    cat(sprintf("[%s] %s batch %d/%d (%d names)...\n",
                format(Sys.time(), "%H:%M:%S"), label, i, n_batches, length(batches[[i]])))

    result <- tryCatch(fetch_genderize_batch(batches[[i]], country_id = country_id),
                        error = function(e) e)

    if (inherits(result, "error")) {
      warning(sprintf("Batch %d/%d failed as a whole (%s) -- retrying its %d names individually",
                       i, n_batches, conditionMessage(result), length(batches[[i]])), call. = FALSE)
      single_results <- lapply(batches[[i]], function(nm) {
        tryCatch(
          fetch_genderize_batch(nm, country_id = country_id, max_tries = 2),
          error = function(e) {
            warning(sprintf("'%s' failed even queried alone (%s) -- skipped, will remain unclassifiable",
                             nm, conditionMessage(e)), call. = FALSE)
            NULL
          }
        )
      })
      single_results <- single_results[!vapply(single_results, is.null, logical(1))]
      if (length(single_results) == 0) next  # every name in this batch was unrecoverable
      result <- bind_rows(single_results)
    }

    batch_df <- result %>%
      select(first_name = name, gender, probability, count) %>%
      mutate(country_id = if (is.null(country_id)) NA_character_ else country_id)
    saveRDS(batch_df, batch_file)
  }

  batch_files <- list.files(checkpoint_dir, full.names = TRUE)
  genderize_cache <- bind_rows(genderize_cache, lapply(batch_files, readRDS))
  saveRDS(genderize_cache, genderize_cache_file)
  unlink(checkpoint_dir, recursive = TRUE)
  invisible(genderize_cache)
}

build_paper_counts_from_edges <- function(net, core_ids_inst) {
  net$edges %>%
    filter(author_id %in% core_ids_inst) %>%
    distinct(work_id, author_id) %>%
    count(author_id, name = "paper_count")
}

build_primary_subfield_from_edges <- function(net, core_ids_inst) {
  net$edges %>%
    filter(author_id %in% core_ids_inst, !is.na(primary_subfield)) %>%
    distinct(work_id, author_id, primary_subfield) %>%
    count(author_id, primary_subfield, name = "n_papers_in_subfield") %>%
    group_by(author_id) %>%
    slice_max(n_papers_in_subfield, n = 1, with_ties = FALSE) %>%
    ungroup() %>%
    select(author_id, primary_subfield)
}

build_model_df_row <- function(net, seniority_w, window_label) {
  data.frame(
    author_id = V(net$g)$name,
    degree    = igraph::degree(net$g),
    strength  = igraph::strength(net$g),
    betweenness = igraph::betweenness(net$g, weights = NA),
    window    = window_label,
    stringsAsFactors = FALSE
  ) %>%
    left_join(seniority_w, by = "author_id")
}

build_model_df_for_windows <- function(inst, nets_by_window, current_years_by_window) {
  core_ids_inst <- core_ids[[inst]]
  author_profiles_inst <- author_profiles[[inst]]
  authors_with_gender <- build_authors_with_gender(author_profiles_inst) %>%
    select(author_id, gender_primary)

  window_labels <- names(nets_by_window)

  seniority_by_window <- setNames(lapply(window_labels, function(w) {
    build_window_seniority(nets_by_window[[w]], author_profiles_inst, core_ids_inst,
                            current_year = current_years_by_window[[w]])
  }), window_labels)

  extras_by_window <- setNames(lapply(window_labels, function(w) {
    build_paper_counts_from_edges(nets_by_window[[w]], core_ids_inst) %>%
      full_join(build_primary_subfield_from_edges(nets_by_window[[w]], core_ids_inst), by = "author_id") %>%
      mutate(window = w)
  }), window_labels)

  model_df <- bind_rows(lapply(window_labels, function(w) {
    build_model_df_row(nets_by_window[[w]], seniority_by_window[[w]], w)
  })) %>%
    mutate(window = factor(window, levels = window_labels)) %>%
    left_join(
      bind_rows(extras_by_window) %>% mutate(window = factor(window, levels = window_labels)),
      by = c("author_id", "window")
    ) %>%
    left_join(authors_with_gender, by = "author_id") %>%
    mutate(
      paper_count = coalesce(paper_count, 0L),
      primary_subfield = case_when(
        paper_count == 0        ~ "no_papers_this_window",
        is.na(primary_subfield) ~ "no_valid_subfield",
        TRUE                     ~ primary_subfield
      )
    )

  top_subfields <- model_df %>%
    filter(primary_subfield != "no_papers_this_window") %>%
    count(primary_subfield, sort = TRUE) %>%
    slice_head(n = 10) %>%
    pull(primary_subfield)

  model_df <- model_df %>%
    mutate(primary_subfield_grp = factor(case_when(
      primary_subfield == "no_papers_this_window" ~ "no_papers_this_window",
      primary_subfield == "no_valid_subfield"      ~ "other_subfield",
      primary_subfield %in% top_subfields          ~ primary_subfield,
      TRUE                                          ~ "other_subfield"
    )))

  model_df %>%
    filter(gender_primary %in% c("male", "female"), paper_count > 0) %>%
    mutate(
      gender_primary = factor(gender_primary, levels = c("male", "female")),
      primary_subfield_grp = droplevels(primary_subfield_grp)
    )
}

check_nb_theta <- function(model, label, threshold = 100) {
  theta <- tryCatch(summary(model)$theta, error = function(e) NA_real_)
  if (is.na(theta)) return(invisible(theta))
  if (theta > threshold) {
    warning(sprintf(
      "%s: theta.ml() likely diverged (theta = %.0f, expected order 1-10 on this project's data) -- distrust this fit's SEs/p-values; prefer the zeroinfl equivalent instead",
      label, theta
    ), call. = FALSE)
  }
  invisible(theta)
}

tiered_convergence_check <- function(model, label) {
  msgs <- tryCatch(model@optinfo$conv$lme4$messages, error = function(e) NULL)
  if (!is.null(msgs)) {
    cat(sprintf("%s: lme4-reported convergence message present (%s)\n", label, paste(msgs, collapse = "; ")))
  }

  relgrad_ok <- tryCatch({
    if (is.null(model@optinfo$derivs)) stop("derivs not populated")
    relgrad <- with(model@optinfo$derivs, solve(Hessian, gradient))
    max_relgrad <- max(abs(relgrad))
    cat(sprintf("%s: Tier 1 -- max relative gradient: %.5f\n", label, max_relgrad))
    max_relgrad < 0.001
  }, error = function(e) {
    cat(sprintf("%s: Tier 1 -- could not compute (%s), escalating to Tier 2\n", label, conditionMessage(e)))
    FALSE
  })
  if (relgrad_ok) {
    cat(sprintf("%s: small relative gradient -- fit is trustworthy, stopping here\n", label))
    return(invisible("clean_gradient"))
  }

  cat(sprintf("%s: Tier 2 -- restarting from final parameters...\n", label))
  ss <- getME(model, c("theta", "fixef"))
  t0 <- Sys.time()
  model_refit <- update(model, start = ss)
  cat(sprintf("  refit done in %.1fs\n", as.numeric(difftime(Sys.time(), t0, units = "secs"))))
  refit_msgs <- tryCatch(model_refit@optinfo$conv$lme4$messages, error = function(e) NULL)
  coef_shift <- max(abs(lme4::fixef(model_refit) - lme4::fixef(model)))
  cat(sprintf("  refit clean: %s | max |coefficient shift|: %.5f\n",
              is.null(refit_msgs), coef_shift))
  if (is.null(refit_msgs) && coef_shift < 0.01) {
    cat(sprintf("%s: restart converged cleanly with a stable coefficient -- treating as resolved\n", label))
    return(invisible("resolved_restart"))
  }

  cat(sprintf("%s: NOT resolved by Tiers 1-2 -- recommend running validate_glmer_across_optimizers() by hand before trusting this result\n", label))
  invisible("needs_manual_allfit")
}

validate_glmer_across_optimizers <- function(model, term) {
  af <- lme4::allFit(model)
  ss <- summary(af)
  list(messages = ss$msgs, estimates = ss$fixef[, term])
}
check_overdispersion <- function(y, label, threshold = 1000) {
  ratio <- var(y) / mean(y)
  cat(sprintf("%s: var/mean = %.1f%s\n", label, ratio,
              if (ratio > threshold) " -- EXTREME, expect glm.nb's theta estimation to be fragile here" else ""))
  invisible(ratio)
}

run_institution_pipeline <- function(inst) {
  cat(sprintf("\n=== %s ===\n", inst))
  net_w <- networks[[inst]]

  model_df_fit <- build_model_df_for_windows(
    inst,
    nets_by_window = list(baseline = net_w$baseline, post_law = net_w$post_law),
    current_years_by_window = list(baseline = 2019, post_law = 2025)
  ) %>%
    mutate(
      m_quotient_z      = as.numeric(scale(m_quotient)),
      academic_age_z    = as.numeric(scale(academic_age)),
      log_paper_count_z = as.numeric(scale(log1p(paper_count)))
    )

  cat(sprintf("%s: model_df_fit rows = %d\n", inst, nrow(model_df_fit)))
  check_overdispersion(model_df_fit$degree, paste(inst, "degree"))

  time_fit <- function(label, expr) {
    t0 <- Sys.time()
    cat(sprintf("[%s] %s: fitting %s...\n", format(t0, "%H:%M:%S"), inst, label))
    result <- expr
    elapsed <- round(as.numeric(difftime(Sys.time(), t0, units = "secs")), 1)
    cat(sprintf("  -> %s done in %ss\n", label, elapsed))
    result
  }

  m_mixed <- time_fit("m_mixed (glmer.nb)", lme4::glmer.nb(
    degree ~ gender_primary * window + m_quotient_z + academic_age_z + log_paper_count_z +
      (1 | author_id) + (1 | primary_subfield_grp),
    data = model_df_fit
  ))
  tiered_convergence_check(m_mixed, paste(inst, "degree glmer.nb"))
  report_icc(m_mixed, paste(inst, "degree glmer.nb"))

  m_betw_lmer <- time_fit("m_betw_lmer", lme4::lmer(
    log1p(betweenness) ~ gender_primary * window + m_quotient_z + academic_age_z + log_paper_count_z +
      (1 | author_id) + (1 | primary_subfield_grp),
    data = model_df_fit
  ))
  tiered_convergence_check(m_betw_lmer, paste(inst, "betweenness lmer"))
  betw_lmer_t <- summary(m_betw_lmer)$coefficients[, "t value"]

  term <- "gender_primaryfemale:windowpost_law"
  comparison <- data.frame(
    institution = inst,
    spec = c("glmer.nb (degree)", "lmer (betweenness, log1p)"),
    estimate = c(
      lme4::fixef(m_mixed)[term],
      lme4::fixef(m_betw_lmer)[term]
    ),
    p_value = c(
      summary(m_mixed)$coefficients[term, "Pr(>|z|)"],
      2 * stats::pnorm(-abs(betw_lmer_t[term]))
    )
  )

  list(
    model_df_fit = model_df_fit,
    models = list(mixed = m_mixed, betw_lmer = m_betw_lmer),
    comparison = comparison
  )
}

compute_assortativity_inputs <- function(net, gender_lookup, subfield_lookup) {
  vids <- V(net$g)$name
  gender_vec   <- gender_lookup[vids]
  subfield_vec <- subfield_lookup[vids]

  keep <- !is.na(gender_vec) & !is.na(subfield_vec)
  g_sub <- induced_subgraph(net$g, vids = which(keep))
  gender_codes <- as.integer(factor(gender_vec[keep], levels = c("male", "female")))

  list(g_sub = g_sub, gender_codes = gender_codes, subfield_strata = subfield_vec[keep])
}

permutation_test_homophily <- function(inputs, n_perm = 2000, seed = 1) {
  set.seed(seed)
  g_sub <- inputs$g_sub; gender_codes <- inputs$gender_codes; strata <- inputs$subfield_strata
  observed <- assortativity_nominal(g_sub, types = gender_codes, directed = FALSE)

  null_dist <- replicate(n_perm, {
    permuted <- gender_codes
    for (s in unique(strata)) {
      idx <- which(strata == s)
      if (length(idx) > 1) permuted[idx] <- sample(gender_codes[idx])
    }
    assortativity_nominal(g_sub, types = permuted, directed = FALSE)
  })

  data.frame(
    n_nodes   = vcount(g_sub),
    observed  = observed,
    null_mean = mean(null_dist),
    null_sd   = sd(null_dist),
    p_value   = mean(abs(null_dist) >= abs(observed))
  )
}

run_community_detection <- function(inst) {
  core_ids_inst <- core_ids[[inst]]
  author_profiles_inst <- author_profiles[[inst]]

  gender_lookup_inst <- build_authors_with_gender(author_profiles_inst)
  gender_lookup_vec  <- setNames(gender_lookup_inst$gender_primary, gender_lookup_inst$author_id)
  gender_lookup_vec[gender_lookup_vec == "unclassifiable"] <- NA

  run_one_window <- function(net, window_label) {
    g_full <- net$g
    comp <- igraph::components(g_full)
    giant_vids <- which(comp$membership == which.max(comp$csize))
    g <- igraph::induced_subgraph(g_full, vids = giant_vids)

    comm <- igraph::cluster_louvain(g, weights = E(g)$weight)

    subfield_df <- build_primary_subfield_from_edges(net, core_ids_inst)
    subfield_lookup <- setNames(subfield_df$primary_subfield, subfield_df$author_id)

    vids <- V(g)$name
    subfield_vec   <- unname(subfield_lookup[vids])
    gender_vec     <- unname(gender_lookup_vec[vids])
    membership_vec <- igraph::membership(comm)

    valid_subfield <- !is.na(subfield_vec)
    nmi <- if (sum(valid_subfield) > 1) {
      igraph::compare(
        as.integer(membership_vec[valid_subfield]),
        as.integer(factor(subfield_vec[valid_subfield])),
        method = "nmi"
      )
    } else NA_real_

    comm_gender <- data.frame(
      author_id = vids,
      community = as.integer(membership_vec),
      gender    = gender_vec,
      stringsAsFactors = FALSE
    ) %>%
      filter(gender %in% c("male", "female")) %>%
      group_by(community) %>%
      mutate(n_in_community = n()) %>%
      ungroup() %>%
      filter(n_in_community >= 10)

    chisq_test_result <- if (nrow(comm_gender) > 0 && length(unique(comm_gender$community)) > 1) {
      tryCatch(chisq.test(table(comm_gender$community, comm_gender$gender)), error = function(e) NULL)
    } else NULL
    chisq_p <- if (!is.null(chisq_test_result)) chisq_test_result$p.value else NA_real_
    cramers_v <- if (!is.null(chisq_test_result)) {
      tbl <- table(comm_gender$community, comm_gender$gender)
      k <- min(nrow(tbl), ncol(tbl))
      sqrt(as.numeric(chisq_test_result$statistic) / (sum(tbl) * (k - 1)))
    } else NA_real_

    gender_by_community <- comm_gender %>%
      group_by(community) %>%
      summarise(n = n(), pct_female = mean(gender == "female"), .groups = "drop") %>%
      arrange(desc(n))

    subfield_by_community <- data.frame(
      community = as.integer(membership_vec),
      subfield  = subfield_vec,
      stringsAsFactors = FALSE
    ) %>%
      filter(!is.na(subfield)) %>%
      count(community, subfield, name = "n_subfield") %>%
      group_by(community) %>%
      mutate(community_n_subfield_classified = sum(n_subfield)) %>%
      slice_max(n_subfield, n = 1, with_ties = FALSE) %>%
      ungroup() %>%
      transmute(
        community,
        dominant_subfield = subfield,
        subfield_purity   = round(n_subfield / community_n_subfield_classified, 3)
      )

    community_profile <- data.frame(community = as.integer(membership_vec)) %>%
      count(community, name = "n_total") %>%
      full_join(gender_by_community %>% select(community, n_gender_classified = n, pct_female), by = "community") %>%
      full_join(subfield_by_community, by = "community") %>%
      arrange(desc(n_total))

    list(
      window              = window_label,
      n_nodes_total        = vcount(g_full),
      n_nodes_giant        = vcount(g),
      giant_comp_share     = round(vcount(g) / vcount(g_full), 3),
      n_communities        = length(comm),
      modularity           = igraph::modularity(comm),
      nmi_vs_subfield      = nmi,
      chisq_p_gender       = chisq_p,
      cramers_v_gender     = cramers_v,
      community_profile    = community_profile
    )
  }

  list(
    baseline = run_one_window(networks[[inst]]$baseline, "baseline"),
    post_law = run_one_window(networks[[inst]]$post_law, "post_law")
  )
}

tidy_model <- function(model, label) {
  one_table <- function(coef_mat, component = NA_character_) {
    data.frame(
      term      = rownames(coef_mat),
      component = component,
      estimate  = coef_mat[, 1],
      std.error = coef_mat[, 2],
      statistic = coef_mat[, 3],
      p.value   = coef_mat[, 4],
      row.names = NULL,
      stringsAsFactors = FALSE
    )
  }

  robust_vcov <- attr(model, "robust_vcov")

  out <- if (!is.null(robust_vcov)) {
    one_table(lmtest::coeftest(model, vcov. = robust_vcov), NA_character_)
  } else if (inherits(model, "zeroinfl") || inherits(model, "hurdle")) {
    s <- summary(model)$coefficients
    rbind(one_table(s$count, "count"), one_table(s$zero, "zero"))
  } else if (inherits(model, "merMod")) {
    cm <- summary(model)$coefficients
    if (ncol(cm) < 4) {
      cm <- cbind(cm, 2 * stats::pnorm(-abs(cm[, 3])))
    }
    one_table(cm, "fixed")
  } else {
    one_table(summary(model)$coefficients, NA_character_)
  }

  out$conf.low  <- out$estimate - 1.96 * out$std.error
  out$conf.high <- out$estimate + 1.96 * out$std.error
  out$model <- label
  out[, c("model", "component", "term", "estimate", "std.error", "statistic",
          "p.value", "conf.low", "conf.high")]
}

model_fit_stats <- function(model, label) {
  n_obs <- if (inherits(model, "merMod")) stats::nobs(model) else length(stats::residuals(model))
  aic_val <- tryCatch(stats::AIC(model), error = function(e) NA_real_)
  theta_val <- if (inherits(model, "merMod")) {
    NA_real_
  } else {
    tryCatch(if (!is.null(model$theta)) model$theta else NA_real_, error = function(e) NA_real_)
  }
  data.frame(model = label, n_obs = n_obs, aic = aic_val, theta = theta_val,
             stringsAsFactors = FALSE)
}
